## View SQL text

```sql
SELECT  
RECIBO.CDUNIECO  
, RECIBO.CDRAMO  
, PRODUTO.dsramo			   					PRODUTO  
, PRODUTO.cdtipram			   					TIPORAMO  
, RECIBO.ESTADO  
, RECIBO.NMPOLIZA  
, APOLICE.estado			   					ESTADO_APOLICE  
, APOLICE.feefecto 								DT_INI_APOLICE  
, DECODE (APOLICE.ottempot, 'R', APOLICE.feproren, 'T', APOLICE.fevencim) 	DT_FIM_APOLICE  
, APOLICE.FEEMISIO								DT_EMISSAO_APOLICE  
, APOLICE.nmcuadro                                                              NMCUADRO  
, APOLICE.CDTIPCOA								CDTIPCOA  
, APOLICE.NMRENOVA								NMRENOVA_APOLICE  
, TIPOS_DOC.ID									CLASSE_ID  
, TIPOS_DOC.DESCRICAO							CLASSE_DOCUMENTO  
, RECIBO.TIPORECI  
, TIPOREC.descripl			   					TIPO_DOCUMENTO  
, RECIBO.NMRECIBO  
, RECIBO.NMRECINUE  
, RECIBO.FEINICIO  
, RECIBO.FEFINAL  
, RECIBO.FEEMISIO  
, RECIBO.CDESTADO  
, RECIBO.FEESTADO  
, RECIBO.CDMOTANU  
, RECIBO.CDGESTOR  
, RECIBO.CDDEVCIA  
, RECIBO.CDAGRUPA  
, RECIBO.CDPERSON  
, DECODE(SIGN(MOVIMENTOS.FEMOVIMI-DECODE(SIGN(RECIBO.FEINICIO-RECIBO.FEEMISIO),-1, RECIBO.FEEMISIO, RECIBO.FEINICIO)), -1, DECODE(SIGN(RECIBO.FEINICIO-RECIBO.FEEMISIO),-1, RECIBO.FEEMISIO, RECIBO.FEINICIO), MOVIMENTOS.FEMOVIMI)  DT_PRODUCAO  
, PKG_FORMULAS_COSEC.GET_DT_VENCIMENTO_RECIBO(RECIBO.TIPORECI,TIPOS_DOC.ID, RECIBO.FEINICIO, RECIBO.FEEMISIO)  DT_VENCIMENTO  
, DECODE(SIGN(MOVIMENTOS.FEMOVIMI-MOVIMENTOS.FEEMISIO), -1, MOVIMENTOS.FEEMISIO, MOVIMENTOS.FEMOVIMI)  DT_ESTADO  
, DECODE(SUBSTR(MOVIMENTOS.CDNEGOCI,1,1),'1','EMITIDO'  
,'2','ANULADO'  
,'3','REABILITADO'  
,DECODE(MOVIMENTOS.CDNEGOCI,'503','ANULADO'  
,'953','EMITIDO'  
,'954','EMITIDO'  
,'955','EMITIDO'  
,'956','EMITIDO'  
,'959','EMITIDO'  
,'***','ANULADO'  
,'DESCONHECIDO')) ESTADO_DOCUMENTO  
, APOLICE.CDMONEDA  
, PKG_FORMULAS_COSEC.FUN_TAXA_CAMBIO(APOLICE.CDMONEDA,'EUR',SYSDATE)			   					TX_CAMBIO_EUR  
, RECIBO.PTIMPORT																TOTAL_RECIBO  
, PKG_FORMULAS_COSEC.FUN_PREMIO_COMERCIAL(RECIBO.CDUNIECO, RECIBO.NMRECIBO)	    PREMIO_COMERCIAL  
, PKG_FORMULAS_COSEC.FUN_BRUTO(RECIBO.CDUNIECO, RECIBO.NMRECIBO)         PREMIO_BRUTO  
, PKG_FORMULAS_COSEC.FUN_PREMIO_COMERCIAL_RP(RECIBO.CDUNIECO, RECIBO.NMRECIBO)	PREMIO_POLITICO_COSEC  
, PKG_FORMULAS_COSEC.FUN_PREMIO_COMERCIAL_RC(RECIBO.CDUNIECO, RECIBO.NMRECIBO)  PREMIO_COMERCIAL_COSEC  
, PKG_FORMULAS_COSEC.FUN_BONIFICACAO(RECIBO.CDUNIECO, RECIBO.NMRECIBO)	        BONIFICACAO_APOLICE  
, PKG_FORMULAS_COSEC.FUN_ACERTO(RECIBO.CDUNIECO, RECIBO.NMRECIBO)	            ACERTO_PREMIO  
, PKG_FORMULAS_COSEC.FUN_CUSTO(RECIBO.CDUNIECO, RECIBO.NMRECIBO)	            APOLICE_ACTA_ADICIONAL  
, PKG_FORMULAS_COSEC.FUN_SELO(RECIBO.CDUNIECO, RECIBO.NMRECIBO)	                SELO_APOLICE  
, PKG_FORMULAS_COSEC.FUN_CUSTO(RECIBO.CDUNIECO, RECIBO.NMRECIBO)	            CUSTO_ADM_SELO_APOLICE  
, PKG_FORMULAS_COSEC.FUN_SELO_GAR(RECIBO.CDUNIECO, RECIBO.NMRECIBO)	            SELO_GARANTIA  
, PKG_FORMULAS_COSEC.FUN_OUTROS_IMP(RECIBO.CDUNIECO, RECIBO.NMRECIBO)	            OUTROS_IMPOSTOS  
, PKG_FORMULAS_COSEC.FUN_TAXA_COMPROMISSO(RECIBO.CDUNIECO, RECIBO.NMRECIBO)	    TAXA_COMPROMISSO  
, PKG_FORMULAS_COSEC.FUN_VALOR_FACTURA(RECIBO.CDUNIECO, RECIBO.NMRECIBO)	    VALOR_FACTURA  
, PKG_FORMULAS_COSEC.FUN_IVA(RECIBO.CDUNIECO, RECIBO.NMRECIBO)		            IVA  
, PKG_FORMULAS_COSEC.FUN_TOTAL_IMPOSTO(RECIBO.CDUNIECO, RECIBO.NMRECIBO)        TOTAL_IMPOSTO  
, PKG_FORMULAS_COSEC.FUN_PARTICIPACAO(RECIBO.CDUNIECO, RECIBO.NMRECIBO)	        PARTICIPACAO_RESULTADOS  
FROM  
GD_TIPOS_DOCUMENTO TIPOS_DOC  
, TRAMOS  PRODUTO  
, TMANTENI TIPOREC  
, MPOLIZAS APOLICE  
, MRECIBO RECIBO  
, tmovirec MOVIMENTOS  
WHERE  
1=1  
--AND ptimport IS NULL  
AND TIPOS_DOC.DATA_FIM >= RECIBO.FEEMISIO  
AND TIPOS_DOC.DATA_INICIO <= RECIBO.FEEMISIO  
AND TIPOS_DOC.SINAL_VALOR=SIGN(RECIBO.PTIMPORT)  
AND TIPOS_DOC.TIPORECI=RECIBO.TIPORECI  
AND TIPOREC.CDTABLA = 'TTIPOREC'  
AND TIPOREC.CODIGO  = RECIBO.TIPORECI  
AND PRODUTO.CDRAMO  = RECIBO.CDRAMO  
AND APOLICE.nmsuplem       = (  
SELECT  
MAX(X.nmsuplem)  
FROM  
mpolizas X  
WHERE  
X.nmpoliza = APOLICE.nmpoliza  
AND X.estado   = APOLICE.estado  
AND X.cdramo   = APOLICE.cdramo  
AND X.cdunieco = APOLICE.cdunieco  
)  
AND APOLICE.nmpoliza       = RECIBO.nmpoliza  
AND APOLICE.estado         = RECIBO.estado  
AND APOLICE.cdramo         = RECIBO.cdramo  
AND APOLICE.cdunieco 	   = RECIBO.cdunieco			/* APOLICE RECIBO */  
AND RECIBO.CDUNIECO = MOVIMENTOS.CDUNIECO  
AND RECIBO.NMRECIBO = MOVIMENTOS.NMRECIBO  
AND DECODE(SUBSTR(MOVIMENTOS.CDNEGOCI,1,1),'1','S'  
,'2','S'  
,'3','S'  
,DECODE(MOVIMENTOS.CDNEGOCI,'503','S'  
,'953','S'  
,'954','S'  
,'955','S'  
,'956','S'  
,'959','S'  
,'***','S'  
,'N')) = 'S'  
ORDER BY RECIBO.NMRECIBO

```
