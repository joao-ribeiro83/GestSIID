## View SQL text

```sql
SELECT  
MOVIMENTO_CONT.CDNMPROC  
,MOVIMENTO_CONT.CDTIPCON  
,MOVIMENTO_CONT.CDUNIECO  
,MOVIMENTO_CONT.CDRAMO  
,MOVIMENTO_CONT.NMPOLIZA  
,MOVIMENTO_CONT.AAAPERTU  
,MOVIMENTO_CONT.NMSINIES  
,MOVIMENTO_CONT.CDPERSON  
,TOMADOR.CDIDEPER  NIPC_TOMADOR  
,TOMADOR.DSNOMBRE  NOME_TOMADOR  
,SUM(DETALHE_PAG.IMPREFER) VALOR_PAGAMENTO  
FROM  
CO_MVCON MOVIMENTO_CONT  
,TRAMOS   RAMOS  
,MPOLIPER PESSOA_APOLICE  
,MPAGOSIN PAGAMENTO  
,MDSPASIN DETALHE_PAG  
,MPERSONA TOMADOR  
WHERE  
MOVIMENTO_CONT.NMORDEN=(SELECT MIN(X.NMORDEN)  
FROM  
CO_MVCON X  
WHERE  
MOVIMENTO_CONT.CDNMPROC=X.CDNMPROC  
AND MOVIMENTO_CONT.CDTIPCON=X.CDTIPCON  
AND MOVIMENTO_CONT.CDUNIECO=X.CDUNIECO  
AND MOVIMENTO_CONT.CDRAMO  =X.CDRAMO  
AND MOVIMENTO_CONT.NMPOLIZA=X.NMPOLIZA  
AND NVL(MOVIMENTO_CONT.AAAPERTU,1)=NVL(X.AAAPERTU,1)  
AND NVL(MOVIMENTO_CONT.NMSINIES,1)=NVL(X.NMSINIES,1)  
)  
AND PAGAMENTO.CDTIPMOV    (+)=1  
AND DETALHE_PAG.CDIMPUES  (+)='INDE'  
AND PAGAMENTO.NMSINIES    (+)= MOVIMENTO_CONT.NMSINIES  
AND PAGAMENTO.STATUS      (+)= MOVIMENTO_CONT.STATUS  
AND PAGAMENTO.AAAPERTU    (+)= MOVIMENTO_CONT.AAAPERTU  
AND PAGAMENTO.CDRAMO      (+)= MOVIMENTO_CONT.CDRAMO  
AND PAGAMENTO.CDUNIECO    (+)= MOVIMENTO_CONT.CDUNIECO  
AND DETALHE_PAG.NMORDPAG  (+)= PAGAMENTO.NMORDPAG  
AND DETALHE_PAG.NMSINIES  (+)= PAGAMENTO.NMSINIES  
AND DETALHE_PAG.STATUS    (+)= PAGAMENTO.STATUS  
AND DETALHE_PAG.AAAPERTU  (+)= PAGAMENTO.AAAPERTU  
AND DETALHE_PAG.CDRAMO    (+)= PAGAMENTO.CDRAMO  
AND DETALHE_PAG.CDUNIECO  (+)= PAGAMENTO.CDUNIECO  
AND PESSOA_APOLICE.CDUNIECO        = MOVIMENTO_CONT.CDUNIECO  
AND PESSOA_APOLICE.CDRAMO          = MOVIMENTO_CONT.CDRAMO  
AND PESSOA_APOLICE.ESTADO          = 'M'  
AND PESSOA_APOLICE.NMPOLIZA        = MOVIMENTO_CONT.NMPOLIZA  
AND PESSOA_APOLICE.NMSITUAC        = 0  
AND PESSOA_APOLICE.CDROL           = DECODE(RAMOS.CDTIPRAM,1,'TO','BE')  
AND PESSOA_APOLICE.NMSUPLEM        = (  
SELECT  
MAX(X.NMSUPLEM)  
FROM  
MPOLIPER X  
WHERE  
X.CDUNIECO = PESSOA_APOLICE.CDUNIECO  
AND   X.CDRAMO   = PESSOA_APOLICE.CDRAMO  
AND   X.ESTADO   = PESSOA_APOLICE.ESTADO  
AND   X.NMPOLIZA = PESSOA_APOLICE.NMPOLIZA  
AND   X.NMSITUAC = PESSOA_APOLICE.NMSITUAC  
AND   X.CDROL    = PESSOA_APOLICE.CDROL  
AND   X.CDPERSON = PESSOA_APOLICE.CDPERSON  
AND   X.NMSUPLEM <= TO_CHAR(SYSDATE,'J')||'99999999999'  
)  
AND TOMADOR.CDPERSON               = PESSOA_APOLICE.CDPERSON  
AND RAMOS.CDRAMO=MOVIMENTO_CONT.CDRAMO  
--AND MOVIMENTO_CONT.CDTIPCON ='CP'  
GROUP BY  
MOVIMENTO_CONT.CDNMPROC  
,MOVIMENTO_CONT.CDTIPCON  
,MOVIMENTO_CONT.CDUNIECO  
,MOVIMENTO_CONT.CDRAMO  
,MOVIMENTO_CONT.NMPOLIZA  
,MOVIMENTO_CONT.AAAPERTU  
,MOVIMENTO_CONT.NMSINIES  
,MOVIMENTO_CONT.CDPERSON  
,TOMADOR.CDIDEPER  
,TOMADOR.DSNOMBRE

```
