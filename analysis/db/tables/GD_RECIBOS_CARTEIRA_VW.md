## View SQL text

```sql
SELECT /*+ LEADING(RECIBO)  */
	RECIBO.cdunieco CDUNIECO
	,RECIBO.CDRAMO CDRAMO
	,PRODUTO.dsramo PRODUTO
	,PRODUTO.cdtipram TIPORAMO
	,RECIBO.CDESTADO CDESTADO
	,ESTADOREC.descripl ESTADO_DOCUMENTO
	,RECIBO.nmpoliza NMPOLIZA
	,RECIBO.TIPORECI TIPODOC_ID
	,TIPOREC.descripl TIPO_DOCUMENTO
	,DIRCOM.otvalor26 DIR_COMERCIAL
	,APOLICE.estado ESTADO_APOLICE
	,APOLICE.feefecto DT_INI_APOLICE
	,APOLICE.nmcuadro NMCUADRO
	,APOLICE.CDTIPCOA CDTIPCOA
	,DECODE(APOLICE.ottempot, 'R', APOLICE.feproren, 'T', APOLICE.fevencim) DT_FIM_APOLICE
	,RECIBO.nmrecibo NMRECIBO
	,RECIBO.nmrecinue N_DOCUMENTO
	,RECIBO.feemisio DT_EMISSAO
	,TOMAPOL.cdperson NUC_TOMADOR
	,TOMAPOL.dsnombre NOME_TOMADOR
	,TOMAPOL.CDTIPIDE TIPO_ID_TOMADOR
	,TOMAPOL.CDIDEPER ID_TOMADOR
	,TOMAPOL.CDPAIS PAIS_TOMADOR
	,APOLICE.cdmoneda MOEDA_APOLICE
	,APOLICE.nmrenova NMRENOVA
	,COMISSAO.CDTIPCOM TIPO_COMISSAO
	,COMISSAO.ptimport COMISSAO
	,AGAPOL.cdperson NUC_INTERVENTOR
	,AGAPOL.cdAgente COD_INTERVENTOR
	,AGAPOL.cdclaage CLASE_INTERVENTOR
	,AGAPOL.NOME_AGENTE NOME_INTERVENTOR
	,RECIBO.ptimport V_CARTEIRA
	,RECIBO.feinicio DT_INICIO_VIG
	,RECIBO.feestado DT_ESTADO
	,RECIBO.fefinal DT_FIN_VIG
	,MAX(TAXA.POCOMPRO) TX_COMISSAO
	,CAMBIO.ptcambio TX_CAMBIO
	,SUM(DECODE(DETALHE_REC.CDTIPCON,'PPDIV',1,0)) furto_ident
	,SUM(DECODE(DETALHE_REC.CDTIPCON, 'BONIF', DETALHE_REC.PTIMPORT, 'PSDPR', DETALHE_REC.PTIMPORT, 'PSDPR ME', DETALHE_REC.PTIMPORT, 'PSDPR MI', DETALHE_REC.PTIMPORT, 'PSDPR PC', DETALHE_REC.PTIMPORT, 'PDPR ME', DETALHE_REC.PTIMPORT, 'PDPR MI', DETALHE_REC.PTIMPORT, 'PDPR PC', DETALHE_REC.PTIMPORT, 'PDPR PE', DETALHE_REC.PTIMPORT, 'PPDIV', DETALHE_REC.PTIMPORT, 0)) PREMIO_COMERCIAL
	,SUM(DECODE(DETALHE_REC.CDTIPCON, 'PDPR PC', DETALHE_REC.PTIMPORT, 'PDPR PE', DETALHE_REC.PTIMPORT, 'PSDPR PC', DETALHE_REC.PTIMPORT, 0)) PREMIO_POLITICO
	,SUM(DECODE(DETALHE_REC.CDTIPCON, 'BONIF', DETALHE_REC.PTIMPORT, 0)) BONIFICACAO_APOLICE
	,SUM(DECODE(DETALHE_REC.CDTIPCON, 'CADM', DETALHE_REC.PTIMPORT, 0)) APOLICE_ACTA_ADICIONAL
	,SUM(DECODE(DETALHE_REC.CDTIPCON, 'ISAPOL', DETALHE_REC.PTIMPORT, 'ISAPPD', DETALHE_REC.PTIMPORT, 0)) SELO_APOLICE
	,SUM(DECODE(DETALHE_REC.CDTIPCON, 'ISCAD', DETALHE_REC.PTIMPORT, 0)) CUSTO_ADM_SELO_APOLICE
	,SUM(DECODE(DETALHE_REC.CDTIPCON, 'ISGAR', DETALHE_REC.PTIMPORT, 0)) SELO_GARANTIA
	,SUM(DECODE(DETALHE_REC.CDTIPCON, 'OUTIMP', DETALHE_REC.PTIMPORT, 'OUTIMPOL', DETALHE_REC.PTIMPORT, 'OUTIMCAD', DETALHE_REC.PTIMPORT, 0)) OUTROS_IMPOSTOS
	,SUM(DECODE(DETALHE_REC.CDTIPCON, 'TXCOMP', DETALHE_REC.PTIMPORT, 0)) TAXA_COMPROMISSO
	,SUM(DECODE(DETALHE_REC.CDTIPCON, 'CAC', DETALHE_REC.PTIMPORT, 'CAP', DETALHE_REC.PTIMPORT, 'CPS', DETALHE_REC.PTIMPORT, 'CVR', DETALHE_REC.PTIMPORT, 0)) VALOR_FACTURA
	,SUM(DECODE(DETALHE_REC.CDTIPCON, 'IVA', DETALHE_REC.PTIMPORT, 0)) IVA
	,SUM(DECODE(DETALHE_REC.CDTIPCON, 'ISAPOL', DETALHE_REC.PTIMPORT, 'ISCAD', DETALHE_REC.PTIMPORT, 'ISGAR', DETALHE_REC.PTIMPORT, 'OUTIMP', DETALHE_REC.PTIMPORT, 'OUTIMPOL', DETALHE_REC.PTIMPORT, 'OUTIMCAD', DETALHE_REC.PTIMPORT, 'IVA', DETALHE_REC.PTIMPORT, 'ISAPPD', DETALHE_REC.PTIMPORT, 0)) TOTAL_IMPOSTOS
	,SUM(DECODE(DETALHE_REC.CDTIPCON, 'PARES', DETALHE_REC.PTIMPORT, 0)) PARTICIPACAO_RESULTADOS
	,SUM(DECODE(DETALHE_REC.CDTIPCON, 'PACERTO', DETALHE_REC.PTIMPORT, 0)) ACERTO_PREMIO
FROM tramos PRODUTO
	,GD_TIPOS_DOCUMENTO TIPOS_REC
	,mparapro ROLETOM
	,TCUCOPRO TAXA
	,tcambios CAMBIO
	,ttapvaat DIRCOM /* */
	,GD_AGENTES_APOLICE_MV AGAPOL /* */
	,tmanteni TIPOREC
	,tmanteni ESTADOREC
	,GD_ATRIBUTOS_APOLICE_MV ATRIBUTOS_APOL /* */
	,GD_INTERVENIENTES_APOLICE_MV TOMAPOL
	,treccom COMISSAO
	,mrecibo RECIBO
	,mpolizas APOLICE
	,mrecidet DETALHE_REC
WHERE PRODUTO.cdramo = RECIBO.cdramo /* RAMO */
	AND comissao.cdunieco(+) = RECIBO.cdunieco
	AND comissao.nmrecibo(+) = RECIBO.nmrecibo /* COMISSAO */
	AND ESTADOREC.cdtabla = 'TESTREC'
	AND ESTADOREC.CODIGO = RECIBO.CDESTADO /* ESTADO DO RECIBO */
	AND TIPOREC.cdtabla = 'TTIPOREC'
	AND TIPOREC.codigo = RECIBO.tiporeci /* TIPO DE RECIBO */
	AND TAXA.NMCUADRO = APOLICE.NMCUADRO
	AND TAXA.CDTIPCON = DETALHE_REC.CDTIPCON
	/* */
	AND dircom.nmtabla = 5
	AND ATRIBUTOS_APOL.ATRIBUTO2 = dircom.otclave1
	AND ATRIBUTOS_APOL.nmpoliza = RECIBO.nmpoliza
	AND ATRIBUTOS_APOL.estado = RECIBO.ESTADO
	AND ATRIBUTOS_APOL.cdramo = RECIBO.cdramo
	AND ATRIBUTOS_APOL.cdunieco = RECIBO.cdunieco /* DIRECCAO COMERCIAL */
	AND roletom.cdramo = RECIBO.cdramo
	AND TOMAPOL.cdrol = roletom.cdroltom
	AND TOMAPOL.nmpoliza = RECIBO.nmpoliza
	AND TOMAPOL.estado = RECIBO.estado
	AND TOMAPOL.cdramo = RECIBO.cdramo
	AND TOMAPOL.cdunieco = RECIBO.cdunieco /* TOMADOR */
	AND DETALHE_REC.nmrecibo = recibo.nmrecibo
	AND DETALHE_REC.cdunieco = recibo.cdunieco /* DETALHES RECIBO */
	AND CAMBIO.fevalor = (
		SELECT MAX(tcam2.fevalor)
		FROM tcambios tcam2
		WHERE tcam2.cdmoneda = CAMBIO.cdmoneda
			AND tcam2.cdmonbas = CAMBIO.cdmonbas
			AND tcam2.fevalor <= GREATEST(RECIBO.FEEMISIO, TO_DATE('01-01-2001', 'DD-MM-RRRR'))
		)
	AND CAMBIO.cdmoneda = (
		SELECT tk.cdeuro
		FROM tkrnlpar tk
		)
	AND CAMBIO.cdmonbas = APOLICE.cdmoneda /* CONVERS?O CAMBIAL */
	AND AGAPOL.nmpoliza = RECIBO.nmpoliza
	AND AGAPOL.estado = RECIBO.estado
	AND AGAPOL.cdramo = RECIBO.cdramo
	AND AGAPOL.cdunieco = RECIBO.cdunieco /* AGENTE APOLICE */
	AND APOLICE.nmsuplem = (
		SELECT MAX(X.nmsuplem)
		FROM mpolizas X
		WHERE X.nmpoliza = APOLICE.nmpoliza
			AND X.estado = APOLICE.estado
			AND X.cdramo = APOLICE.cdramo
			AND X.cdunieco = APOLICE.cdunieco
		)
	AND APOLICE.nmpoliza = RECIBO.nmpoliza
	AND APOLICE.estado = RECIBO.estado
	AND APOLICE.cdramo = RECIBO.cdramo
	AND APOLICE.cdunieco = RECIBO.cdunieco /* APOLICE RECIBO */
	AND DETALHE_REC.CDTIPCON != 'COMPS'
	AND recibo.tiporeci = TIPOS_REC.TIPORECI
	AND DECODE(TIPOS_REC.ID, 1, 'S', 2, 'S', 6, 'S', 'N') = 'S'
GROUP BY RECIBO.cdunieco
	,RECIBO.feestado
	,RECIBO.TIPORECI
	,TIPOREC.descripl
	,RECIBO.nmrecibo
	,RECIBO.nmrecinue
	,RECIBO.feinicio
	,RECIBO.feemisio
	,RECIBO.FEFINAL
	,PRODUTO.dsramo
	,PRODUTO.cdtipram
	,RECIBO.nmpoliza
	,TOMAPOL.cdperson
	,TOMAPOL.dsnombre
	,TOMAPOL.CDPAIS
	,APOLICE.estado
	,APOLICE.cdmoneda
	,APOLICE.feefecto
	,APOLICE.nmrenova
	,APOLICE.nmcuadro
	,APOLICE.CDTIPCOA
	,DECODE(APOLICE.ottempot, 'R', APOLICE.feproren, 'T', APOLICE.fevencim)
	,ESTADOREC.descripl
	,CAMBIO.ptcambio
	,COMISSAO.CDTIPCOM
	,COMISSAO.ptimport
	,RECIBO.CDRAMO /* */
	,RECIBO.CDESTADO
	,DIRCOM.otvalor26
	,TOMAPOL.CDTIPIDE
	,TOMAPOL.CDIDEPER
	,AGAPOL.cdperson
	,AGAPOL.CDAGENTE
	,AGAPOL.cdclaage
	,AGAPOL.NOME_AGENTE
	,RECIBO.ptimport
```
