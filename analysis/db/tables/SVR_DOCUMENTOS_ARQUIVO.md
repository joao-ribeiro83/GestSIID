## View SQL text

```sql
SELECT /*+ ordered USE_NL(B C)*/
	B.ID DOCUMENTO_ID
	,B.modelo_id MODELO_ID
	,C.DESCRICAO
	,C.TIPO_PARAMETRO
	,C.NOME_PARAMETRO
	,C.ORDEM_PARAMETRO
	,CASE C.N_ATRIBUTO
		WHEN 0
			THEN C.VALOR_OMISSAO
		WHEN 1
			THEN B.ATRIB_ARQ_1
		WHEN 2
			THEN B.ATRIB_ARQ_2
		WHEN 3
			THEN B.ATRIB_ARQ_3
		WHEN 4
			THEN B.ATRIB_ARQ_4
		WHEN 5
			THEN B.ATRIB_ARQ_5
		WHEN 6
			THEN B.ATRIB_ARQ_6
		WHEN 7
			THEN B.ATRIB_ARQ_7
		WHEN 8
			THEN B.ATRIB_ARQ_8
		WHEN 9
			THEN B.ATRIB_ARQ_9
		WHEN 10
			THEN B.ATRIB_ARQ_10
		WHEN 11
			THEN B.ATRIB_ARQ_11
		WHEN 12
			THEN B.ATRIB_ARQ_12
		WHEN 13
			THEN B.ATRIB_ARQ_13
		WHEN 14
			THEN B.ATRIB_ARQ_14
		WHEN 15
			THEN B.ATRIB_ARQ_15
		WHEN 16
			THEN B.ATRIB_ARQ_16
		WHEN 17
			THEN B.ATRIB_ARQ_17
		WHEN 18
			THEN B.ATRIB_ARQ_18
		WHEN 19
			THEN B.ATRIB_ARQ_19
		WHEN 20
			THEN B.ATRIB_ARQ_20
		ELSE NULL
		END VALOR_ATRIBUTO
	,B.nome_output NOME_FICHEIRO
FROM SVR_DOCUMENTOS B
	,DOC_ATRIBUTOS_ARQUIVO C
WHERE 1 = 1
	AND B.MODELO_ID = C.MODELO_ID
	AND B.ARQ_ID = C.ARQ_ID
	AND B.data_pedido BETWEEN C.data_inicio
	AND C.data_fim
```
