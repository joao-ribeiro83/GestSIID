## View SQL text

```sql
SELECT /*+ LEADING(VIGIADA) USE_NL(FONTE) */
  VIGIADA.CDFUENTE
, FONTE.DSFUENTE         DESCRICAO
, FONTE.SWPEDIDO
, FONTE.TIPCONVA
, FONTE.TIPFUEN
, FONTE.CDMERCATO
, FONTE.CDCOSEC
, DADOS_FONTE.CDPERSON   NUC_FONTE
, DADOS_FONTE.CDTIPIDE
, DADOS_FONTE.CDIDEPER
, DADOS_FONTE.DSNOMBRE   NOME_FONTE
, DADOS_FONTE.CDPAIS     PAIS_FONTE
, VIGIADA.FEHASTA        DATA_FIM
FROM
  CO_HENTISUR VIGIADA
, CO_FUENTES  FONTE
, MPERSONA    DADOS_FONTE
WHERE DADOS_FONTE.CDPERSON = FONTE.CDPERSON
AND   FONTE.CDFUENTE       = VIGIADA.CDFUENTE
AND   NOT EXISTS (SELECT *
                  FROM   CO_HENTISUR A
                  WHERE  A.CDPERSON = VIGIADA.CDPERSON
                  AND    A.CDFUENTE = VIGIADA.CDFUENTE
                  AND    A.FEHASTA IS NULL)
AND
  (''''||VIGIADA.CDFUENTE||'''',''''||TO_CHAR(VIGIADA.FEHASTA,'DD-MM-YYYY')||'''') NOT IN
    (SELECT
       A.PARAMETRO04
     , A.PARAMETRO03
     FROM
       SVR_DOCUMENTOS           A
     WHERE
       A.MODELO_ID = 'D12.J10')
group by
  VIGIADA.CDFUENTE
, FONTE.DSFUENTE
, FONTE.SWPEDIDO
, FONTE.TIPCONVA
, FONTE.TIPFUEN
, FONTE.CDMERCATO
, FONTE.CDCOSEC
, DADOS_FONTE.CDPERSON
, DADOS_FONTE.CDTIPIDE
, DADOS_FONTE.CDIDEPER
, DADOS_FONTE.DSNOMBRE
, DADOS_FONTE.CDPAIS
, VIGIADA.FEHASTA
```
