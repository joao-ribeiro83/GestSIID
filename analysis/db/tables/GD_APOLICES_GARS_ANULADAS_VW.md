## View SQL text

```sql
SELECT /*+ FIRST_ROWS LEADING(SUP) USE_NL(SUPLEMENTO) USE_NL(RAMO) USE_NL(APOLICE) USE_NL(PESSOAS_APOLICE)*/
	SUP.NMPOLIZA
	,SUP.CDRAMO
	,SUP.CDUNIECO
	,APOLICE.FEEFECTO
	,DECODE(APOLICE.OTTEMPOT, 'R', TRUNC(APOLICE.FEPROREN) - 1, 'T', APOLICE.FEVENCIM, NULL) FEPROREN
	,PESSOA.DSNOMBRE TOMADOR
	,SUP.FEEMISIO FEINIVAL
	,situacoes.nmsituac
FROM TRAMOS RAMO
	,MPERSONA PESSOA
	,MPOLIPER PESSOAS_APOLICE
	,MPOLIZAS APOLICE
	,MPOLISIT SITUACOES
	,MSUPLEME SUPLEMENTO
	,TDESCSUP SUP
WHERE RAMO.CDTIPORA = 1
	AND RAMO.CDRAMO = SITUACOES.CDRAMO
	AND PESSOA.CDPERSON = PESSOAS_APOLICE.CDPERSON
	AND PESSOAS_APOLICE.CDROL = 'TO'
	AND PESSOAS_APOLICE.NMSITUAC = 0
	AND PESSOAS_APOLICE.NMPOLIZA = APOLICE.NMPOLIZA
	AND PESSOAS_APOLICE.CDRAMO = APOLICE.CDRAMO
	AND PESSOAS_APOLICE.CDUNIECO = APOLICE.CDUNIECO
	AND PESSOAS_APOLICE.ESTADO = 'M'
	AND PESSOAS_APOLICE.NMSUPLEM = (
		SELECT MAX(MPOLIPERA.NMSUPLEM)
		FROM MPOLIPER MPOLIPERA
		WHERE MPOLIPERA.CDROL = PESSOAS_APOLICE.CDROL
			AND MPOLIPERA.NMSUPLEM <= TO_CHAR(SYSDATE, 'J') || '99999999999'
			AND MPOLIPERA.NMPOLIZA = PESSOAS_APOLICE.NMPOLIZA
			AND MPOLIPERA.ESTADO = PESSOAS_APOLICE.ESTADO
			AND MPOLIPERA.CDRAMO = PESSOAS_APOLICE.CDRAMO
			AND MPOLIPERA.CDUNIECO = PESSOAS_APOLICE.CDUNIECO
		)
	AND APOLICE.CDUNIECO = SITUACOES.CDUNIECO
	AND APOLICE.CDRAMO = SITUACOES.CDRAMO
	AND APOLICE.NMPOLIZA = SITUACOES.NMPOLIZA
	AND APOLICE.ESTADO = 'M'
	AND APOLICE.CDTIPCOA != 'A'
	AND APOLICE.NMSUPLEM = (
		SELECT MAX(A.NMSUPLEM)
		FROM MPOLIZAS A
		WHERE A.NMSUPLEM <= TO_CHAR(SYSDATE, 'J') || '99999999999'
			AND A.NMPOLIZA = APOLICE.NMPOLIZA
			AND A.ESTADO = APOLICE.ESTADO
			AND A.CDRAMO = APOLICE.CDRAMO
			AND A.CDUNIECO = APOLICE.CDUNIECO
		)
	AND NOT EXISTS (
		SELECT 1
		FROM CO_TRANSFGAR X
		WHERE X.NMPOLIZA = SITUACOES.NMPOLIZA
			AND X.ESTADO = SITUACOES.ESTADO
			AND X.CDRAMO = SITUACOES.CDRAMO
			AND X.CDUNIECO = SITUACOES.CDUNIECO
		)
	AND SITUACOES.NMSUPLEM = SUPLEMENTO.NMSUPLEM
	AND SITUACOES.NMPOLIZA = SUPLEMENTO.NMPOLIZA
	AND SITUACOES.ESTADO = SUPLEMENTO.ESTADO
	AND SITUACOES.CDRAMO = SUPLEMENTO.CDRAMO
	AND SITUACOES.CDUNIECO = SUPLEMENTO.CDUNIECO
	AND SUPLEMENTO.NSUPLOGI = SUP.NSUPLOGI
	AND SUPLEMENTO.NMPOLIZA = SUP.NMPOLIZA
	AND SUPLEMENTO.ESTADO = SUP.ESTADO
	AND SUPLEMENTO.CDRAMO = SUP.CDRAMO
	AND SUPLEMENTO.CDUNIECO = SUP.CDUNIECO
	/* 2003/12/05  
** NFarinha  
AND DECODE(sup.cdtipsup, 55, 'S', 71, 'S', 'N') = 'S'*/
	--AND DECODE(sup.cdtipsup, 9, 'S', 55, 'S', 71, 'S', 'N') = 'S'  
	AND sup.cdtipsup IN (
		9
		,55
		,71
		)
	/* Fim  
** 2003/12/05 */
	AND SUP.ESTADO = 'M'
```
