SELECT  
NOTA_CREDITO.CDUNIECO

, NOTA_CREDITO.NMPOLIZA

, NOTA_CREDITO.FEINICIO  
, NOTA_CREDITO.FEFINAL  
, NOTA_CREDITO.FEEMISIO  
, NOTA_CREDITO.CDESTADO  
, NOTA_CREDITO.FEESTADO  
, NOTA_CREDITO.PTIMPORT

, NOTA_CREDITO.CDDEVCIA

, NOTA_CREDITO.SWFINANC  
, NOTA_CREDITO.CDAGRUPA  
, NOTA_CREDITO.NMRECINUE  
, NOTA_CREDITO.CDPERSON  
, MAX(DECODE(FACTURA.CDATRIBU, 2, FACTURA.OTVALOR, NULL))        TIPOFACTURA  
, MAX(DECODE(FACTURA.CDATRIBU, 1, FACTURA.OTVALOR, NULL))        NMFACTURA  
FROM  
CO_TVALOREC FACTURA  
, MRECIBO     NOTA_CREDITO  
WHERE  
FACTURA.NMRECIBO = NOTA_CREDITO.NMRECIBO  
AND FACTURA.CDUNIECO = NOTA_CREDITO.CDUNIECO  
AND DECODE(NOTA_CREDITO.TIPORECI, 84, 'S'  
, 85, 'S'  
, 86, 'S'  
, 87, 'S'  
, 88, 'S'  
, 89, 'S'  
, 'N') = 'S'  
GROUP BY  
NOTA_CREDITO.CDUNIECO  
, NOTA_CREDITO.CDRAMO  
, NOTA_CREDITO.ESTADO  
, NOTA_CREDITO.NMPOLIZA  
, NOTA_CREDITO.NMSUPLEM
