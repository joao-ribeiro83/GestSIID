SELECT DOC.ID DOC
	,Docs.ID DOC_ID
	,ANEXO.ID ANEXO
	,Anexos.ID ANEXO_ID
	,DOC.ID || '-' || DOC.DESCRICAO || CHR(10) || ANEXO.ID || '-' || ANEXO.DESCRICAO DOCDESC
	,'N' REGISTADA
	,Docs.N_REFERENCIA N_REFERENCIA
	,Docs.DESTINATARIO DESTINATARIO
	,Docs.DATA_IMPRESSAO DATA_IMPRESSAO
	,Docs.DATA_AUDITORIA DATA_AUDITORIA
	,Docs.N_FOLHAS DOC_NFOLHAS
	,Anexos.N_FOLHAS ANEXO_NFOLHAS
FROM DOC_MODELOS_DOCUMENTO DOC
	,DOC_MODELOS_DOCUMENTO ANEXO
	,SVR_DOCUMENTOS Docs
	,SVR_DOCUMENTOS Anexos
WHERE Anexos.MODELO_ID = ANEXO.ID
	AND Docs.MODELO_ID = DOC.ID
	AND Docs.LOTE_ID = Anexos.LOTE_ID
	AND Docs.LOTE_ORDEM < Anexos.LOTE_ORDEM
	AND NOT EXISTS (
		SELECT 1
		FROM SVR_DOCUMENTOS X
		WHERE X.ID > DOCS.ID
			AND X.LOTE_ID = DOCS.LOTE_ID
			AND X.LOTE_ORDEM < ANEXOS.LOTE_ORDEM
		)
	-- AND Docs.N_FOLHAS != 0
	-- AND Anexos.N_FOLHAS != 0
	AND Docs.N_FOLHAS > 0.5
	AND Anexos.N_FOLHAS > 0.5
	AND Docs.IMPRESSORA_ID IN (
		SELECT CHAVE
		FROM CFG_VALORES_DOMINIO
		WHERE DOMINIO_ID = 'IMPRESSORAS_AUDITADAS'
			AND sysdate BETWEEN DATA_INICIO
				AND NVL(DATA_FIM, sysdate)
		)
	AND Docs.MODELO_ID = 'D1.A5'
	AND Anexos.MODELO_ID = 'R3.D27'
--AND DECODE(INSTR(docs.n_referencia,'/',1,2),0,'1',SUBSTR(docs.n_referencia,INSTR(docs.n_referencia,'/',1,2)+1)) = '1'

UNION ALL
