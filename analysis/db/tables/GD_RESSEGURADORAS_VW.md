## View SQL text

```sql
SELECT  
COMPANHIA.CDCIA          CODIGO_RESSEGURADORA  
, ENTIDADE.CDPERSON        NUC_ENTIDADE  
, DECODE(ENTIDADE.CDTIPIDE,0,'NUC/','NIF/')||ENTIDADE.CDIDEPER         ID_ENTIDADE  
, ENTIDADE.DSNOMBRE         NOME_ENTIDADE  
, DOMICILIO.CDTIPDOM  
, DOMICILIO.DSDOMICI        MORADA  
, DECODE(TRIM(DOMICILIO.CDPOSTAL), null, DOMICILIO.OTPISO  
, SUBSTR(DOMICILIO.CDPOSTAL,1,4)||'-'||SUBSTR(DOMICILIO.CDPOSTAL,5,7)||' '||CODIGO_POSTAL.DSCODPOS)  CODIGO_POSTAL  
, ENTIDADE.CDPAIS           CODIGO_PAIS  
, TRIM(PAISES.DESCRIPC)     PAIS_ENTIDADE  
, NATUREZA.DSENG            NATUREZA_JURIDICA  
--, TRUNC((TO_CHAR(LIQUIDACAO.FEPRELIQ,'MM')+2)/3)   TRIMESTRE  
--, TO_CHAR(LIQUIDACAO.FEPRELIQ,'YYYY')              ANO_EXERCICIO  
--, LIQUIDACAO.MODALI                                ANO_SUBSCRICAO  
--, /*NVL(SUM(*/NVL(LIQUIDACAO.PTIMPORT,0)/*),0)*/           CONTABILIDADE  
FROM  
TCIAS COMPANHIA  
, CO_NATUJUR  NATUREZA  
, TMANTENI  PAISES  
, TCODIPOS CODIGO_POSTAL  
, MPERSONA ENTIDADE  
, MDOMICIL DOMICILIO  
--, MPRELIQR  LIQUIDACAO  
WHERE  
1=1  
AND COMPANHIA.CDPERSON         = ENTIDADE.CDPERSON  
AND DOMICILIO.CDPERSON         = COMPANHIA.CDPERSON  
AND DOMICILIO.NMORDDOM         = COMPANHIA.NMORDDOM  
AND CODIGO_POSTAL.CDPAIS   (+) = DOMICILIO.CDPAIS  
AND CODIGO_POSTAL.CDCODPOS (+) = DOMICILIO.CDPOSTAL  
AND PAISES.codigo          (+) = DOMICILIO.CDPAIS  
AND PAISES.CDTABLA         (+) = 'TPAISES'  
AND ENTIDADE.CDNATJUR          = NATUREZA.CDNATJUR  
--AND COMPANHIA.CDCIA            = LIQUIDACAO.CDCIA  
--AND DECODE(LIQUIDACAO.CDCONLIQ,201,MOD(TO_CHAR(LIQUIDACAO.FEPRELIQ,'MM'),3),0)=0  
--AND LIQUIDACAO.CDCONLIQ = 800

```
