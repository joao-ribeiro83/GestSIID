		,DECODE(RECIBO.cdestado, 1, RECIBO.FEESTADO, 5, RECIBO.FEESTADO, 13, RECIBO.FEESTADO, 16, RECIBO.FEESTADO, 17, RECIBO.FEESTADO, 6, RECIBO.FEEMISIO, DECODE(dual.DUMMY, '2', GREATEST(RECIBO.FEESTADO, RECIBO.FEEMISIO, RECIBO.FEINICIO), GREATEST(RECIBO.FEINICIO, RECIBO.FEEMISIO)))
