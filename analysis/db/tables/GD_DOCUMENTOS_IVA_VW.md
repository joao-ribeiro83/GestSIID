## View SQL text

```sql
SELECT  
DOCUMENTO.CDUNIECO  
, DOCUMENTO.TIPORECI  
, PKG_FORMULAS_COSEC.FUN_TIPOAVISO(DOCUMENTO.CDUNIECO, DOCUMENTO.NMRECIBO) TIPO_DOC  
, DOCUMENTO.FEINICIO  
, DOCUMENTO.FEFINAL  
, DOCUMENTO.NMRECIBO  
, DOCUMENTO.NMRECINUE  
, DOCUMENTO.CDRAMO  
, DOCUMENTO.ESTADO  
, DOCUMENTO.NMPOLIZA  
, DOCUMENTO.NMSUPLEM  
, SUM(DECODE(DETALHE.CDTIPCON, 'IVA', 0  
, DETALHE.PTIMPORT))		VALOR_TOTAL  
, SUM(DECODE(DETALHE.CDTIPCON, 'IVA', DETALHE.PTIMPORT  
, 0)) 				VALOR_IVA  
FROM  
MRECIBO  DOCUMENTO  
, MRECIDET DETALHE  
WHERE  
DETALHE.NMRECIBO = DOCUMENTO.NMRECIBO  
AND DETALHE.CDUNIECO = DOCUMENTO.CDUNIECO  
AND DECODE(DOCUMENTO.TIPORECI, 81, 'S'  
, 82, 'S'  
, 83, 'S'  
, 84, 'S'  
, 85, 'S'  
, 86, 'S'  
, 87, 'S'  
, 88, 'S'  
, 89, 'S'  
, 90, 'S'  
, 91, 'S'  
, 92, 'S'  
, 95, 'S'  
, 96, 'S'  
, 97, 'S'  
, 'N') = 'S'  
GROUP BY  
DOCUMENTO.CDUNIECO  
, DOCUMENTO.TIPORECI  
, PKG_FORMULAS_COSEC.FUN_TIPOAVISO(DOCUMENTO.CDUNIECO, DOCUMENTO.NMRECIBO)  
, DOCUMENTO.FEINICIO  
, DOCUMENTO.FEFINAL  
, DOCUMENTO.NMRECIBO  
, DOCUMENTO.NMRECINUE  
, DOCUMENTO.CDRAMO  
, DOCUMENTO.ESTADO  
, DOCUMENTO.NMPOLIZA  
, DOCUMENTO.NMSUPLEM

```
